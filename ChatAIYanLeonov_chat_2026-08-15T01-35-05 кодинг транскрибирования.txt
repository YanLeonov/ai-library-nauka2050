r'--psm 6 '
r'-c load_system_dawg=0 '
r'-c load_freq_dawg=0 '

r'-c load_number_dawg=0'

threshold = 127
img = img.point(lambda p: 255 if p > threshold else 0)
return img

# Передаем путь К ПАПКЕ С ФАЙЛАМИ через двойные обратные слэши Windows

r'--tessdata-dir "C:\\TransKrib\\tessdata" '

r'-c load_punc_dawg=0 '

import os
import pytesseract
from PIL import Image, ImageEnhance, ImageFilter
# Жесткая привязка пути к самой программе
pytesseract.pytesseract.tesseract_cmd = r'C:\Program Files\Tesseract-OCR\tesseract.exe'
def preprocess_image(image_path):
""" Предобработка: очистка картинки для считывания чистой геометрии контуров """
img = Image.open(image_path).convert('L')
enhancer = ImageEnhance.Contrast(img)
img = enhancer.enhance(3.0)
img = img.filter(ImageFilter.SHARPEN)
threshold = 127
img = img.point(lambda p: 255 if p > threshold else 0)
return img
def diplomatic_ocr(image_path):
cleaned_img = preprocess_image(image_path)
# Полный дефолт: Tesseract сам всё возьмет из своей родной папки Program Files
custom_config = (
r'--psm 6 '
r'-c load_system_dawg=0 '
r'-c load_freq_dawg=0 '
r'-c load_punc_dawg=0 '
r'-c load_number_dawg=0'
)
# Имя языка rus совпадает с перенесенным файлом rus.traineddata
return pytesseract.image_to_string(cleaned_img, lang='rus', config=custom_config)
if __name__ == "__main__":
files = [f for f in os.listdir('.') if f.lower().endswith(('.png', '.jpg', '.jpeg'))]
if not files:
print("Ошибка: закинь скриншоты страниц книги в эту папку!")
else:
files.sort()
print(f"Найдено файлов для обработки: {len(files)}")
with open("all_book_text.txt", "w", encoding="utf-8") as total_file:
for idx, target_image in enumerate(files, 1):
print(f"[{idx}/{len(files)}] Обрабатываю файл: {target_image}...")
try:
result_text = diplomatic_ocr(target_image)
total_file.write(result_text)
except Exception as e:
print(f"Ошибка при обработке файла {target_image}: {e}")
print("\n[Успешно] Все страницы обработаны! Результат сохранен в файл: all_book_text.txt")
