Для решения проблемы с путями и пробелами в Windows необходимо передать правильный параметр конфигурации --tessdata-dir в сам движок Tesseract через pytesseract.
Поскольку в Windows пробел в пути Program Files часто ломает стандартное экранирование строк, самым надежным решением является использование короткого имени DOS (8.3) для этой папки — PROGRA~1, либо явное обертывание пути в экранированные двойные кавычки.
Вот скорректированная функция diplomatic_ocr и пример ее интеграции:
python
import os
import subprocess
import pytesseract
from PIL import Image
# 1. Явно указываем путь к исполняемому файлу Tesseract
pytesseract.pytesseract.tesseract_cmd = r"C:\Program Files\Tesseract-OCR\tesseract.exe"
def diplomatic_ocr(image_path):
"""Функция побуквенного распознавания с жестким указанием пути к tessdata
и отключением встроенных словарей.
"""
# Используем путь в формате DOS (8.3), чтобы гарантированно избежать проблем с пробелами в Windows
tessdata_dir = r"C:\PROGRA~1\Tesseract-OCR\tessdata"
# Параметры конфигурации:
# --tessdata-dir принудительно задает папку с моделями
# --psm 6 (Assume a single uniform block of text) хорошо подходит для страниц книг
# load_system_dawg=0 и load_freq_dawg=0 полностью отключают словари Tesseract
custom_config = (
f'--tessdata-dir "{tessdata_dir}" '
f"-l rus "
f"--psm 6 "
f"-c load_system_dawg=0 "
f"-c load_freq_dawg=0"
)
try:
# Открываем изображение и отправляем в Tesseract
img = Image.open(image_path)
text = pytesseract.image_to_string(img, config=custom_config)
return text
except Exception as e:
print(f"Ошибка при обработке {image_path}: {e}")
return ""
# Пример цикла обработки для вашей структуры папок
if __name__ == "__main__":
base_dir = r"C:\TransKrib"
output_file = os.path.join(base_dir, "all_book_text.txt")
# Очищаем или создаем файл перед записью
with open(output_file, "w", encoding="utf-8") as f:
f.write("")
# Обрабатываем файлы 001.png – 005.png
for i in range(1, 6):
file_name = f"{i:03d}.png"  # Форматирует в 001.png, 002.png и т.д.
file_path = os.path.join(base_dir, file_name)
if os.path.exists(file_path):
print(f"Распознавание файла: {file_name}...")
parsed_text = diplomatic_ocr(file_path)
# Записываем результат в общий файл
with open(output_file, "a", encoding="utf-8") as f:
f.write(f"\n\n--- {file_name} ---\n\n")
f.write(parsed_text)
else:
print(f"Файл не найден: {file_path}")
print(f"Готово! Все тексты сохранены в {output_file}")
Используйте код с осторожностью.
Что изменилось и почему это сработает:
Путь C:\PROGRA~1: Windows на системном уровне преобразует Program Files в этот короткий формат. Это полностью устраняет баг, когда Tesseract спотыкался на знаке пробела и искал файлы в папке запуска скрипта.
Конфигурация --tessdata-dir: Мы передаем путь к языковым моделям не через переменные окружения (которые Tesseract в Windows часто игнорирует), а напрямую в параметрах вызова движка.

def preprocess_image(image_path):
"""Функция Python для очистки скриншота от шумов перед отправкой в ИИ"""
print(f"Очистка изображения от шумов: {os.path.basename(image_path)}...")
img = Image.open(image_path).convert('L') # Переводим в черно-белый формат
# Повышаем контрастность, чтобы убрать серые шумы фона
enhancer = ImageEnhance.Contrast(img)
img = enhancer.enhance(2.5)
# Слегка сглаживаем мелкие пиксельные шумы матрицы
img = img.filter(ImageFilter.SMOOTH_MORE)
# Сохраняем временный очищенный файл
temp_path = image_path.replace(".png", "_clean.png")
img.save(temp_path)
return temp_path

print(f"Отправка в ИИ на дипломатическое транскрибирование...")
img = Image.open(clean_image_path)
# Используем модель gemini-2.5-flash, она идеально и быстро работает с картинками
response = client.models.generate_content(
model='gemini-2.5-flash',
contents=[img, "Выполни дипломатическую транскрипцию этой страницы по инструкции."],
config=types.GenerateContentConfig(
system_instruction=SYSTEM_INSTRUCTION,
temperature=0.0 # Минимальный показатель, чтобы ИИ не фантазировал, а строго сканировал
)
return response.text

with open(output_file, "w", encoding="utf-8") as f:

for img_path in images:

except Exception as e:

finally:
# Удаляем временный очищенный файл, чтобы не засорять папку

def preprocess_image(image_path):
"""Очистка скриншота от шумов (бинаризация и контраст) перед отправкой в ИИ"""
print(f"Очистка изображения от шумов: {os.path.basename(image_path)}...")
img = Image.open(image_path).convert('L') # В градации серого
# Повышаем контрастность, убирая серый фон и делая буквы четче
enhancer = ImageEnhance.Contrast(img)
img = enhancer.enhance(2.5)
img = img.filter(ImageFilter.SMOOTH_MORE)
# Сохраняем временный файл
temp_path = image_path.replace(".png", "_clean.png")
img.save(temp_path)
return temp_path
def ask_gemini_to_transcribe(clean_image_path):
"""Прямая отправка очищенного скриншота в ИИ Google Gemini с вашей инструкцией"""
print(f"Отправка в ИИ на дипломатическое транскрибирование...")
img = Image.open(clean_image_path)
# Модель gemini-2.5-flash идеально работает с изображениями
response = client.models.generate_content(
model='gemini-2.5-flash',
contents=[img, "Выполни дипломатическую транскрипцию этой страницы по инструкции."],
config=types.GenerateContentConfig(
system_instruction=SYSTEM_INSTRUCTION,
temperature=0.0 # Ноль, чтобы ИИ строго сканировал и не фантазировал
)
return response.text

with open(output_file, "w", encoding="utf-8") as f:

for img_path in images:

# 2. Отправка в ИИ и получение транскрибированного по правилам текста

with open(output_file, "a", encoding="utf-8") as f:
f.write(final_text)

# Удаляем временную копию
if os.path.exists(temp_clean_img):
os.remove(temp_clean_img)
print("\n[Всё] Программа успешно завершила работу! Проверьте файл all_book_text.txt.")

Вариант А: Перейти на бесплатное API от другого разработчика (например, OpenAI / ChatGPT или Anthropic / Claude), у которых бесплатные лимиты на чтение картинок через прокси часто лояльнее.
Вариант Б: Использовать локальную нейросеть (через программу Ollama), которая будет работать прямо на вашем компьютере, вообще не требуя интернета, прокси, VPN и ключей, но ей нужна хорошая видеокарта.
