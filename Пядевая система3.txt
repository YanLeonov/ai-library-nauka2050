function buildTables() {
 const sTable = document.getElementById('small-table');
 const mTable = document.getElementById('medium-table');
 const b2Table = document.getElementById('body2-table');
 sTable.innerHTML = ""; mTable.innerHTML = ""; b2Table.innerHTML = "";

 measures.small.forEach(m => {
 const tr = document.createElement('tr'); tr.className = 'clickable-row'; tr.id = `m-${m.id}`;
 tr.innerHTML = `<td><strong>${m.sign}</strong></td><td>${m.соотв}</td><td>${m.val}</td><td>${m.name}</td>`;
 tr.onclick = () => selectMeasure(m); sTable.appendChild(tr);
 });

 measures.medium.forEach(m => {
 const tr = document.createElement('tr'); tr.className = 'clickable-row'; tr.id = `m-${m.id}`;
 tr.innerHTML = `<td><strong>${m.sign}</strong></td><td>${m.соотв}</td><td>${m.val}</td><td>${m.name}</td>`;
 tr.onclick = () => selectMeasure(m); mTable.appendChild(tr);
 });

 measures.body2.forEach(m => {
 const tr = document.createElement('tr'); tr.className = 'clickable-row'; tr.id = `m-${m.id}`;
 tr.innerHTML = `<td><strong>${m.sign}</strong></td><td>${m.name}</td>`;
 tr.onclick = () => selectMeasure(m); b2Table.appendChild(tr);
 });
}

const canvas = document.getElementById('paintCanvas');
const ctx = canvas.getContext('2d');
let activeImgType = "body";
let activeMeasure = null;
let activeStep = 1;
let pulseValue = 1.0;
let pulseDir = -0.03;

function selectMeasure(m) {
 activeStep = null; activeMeasure = m; activeImgType = m.file;
 document.querySelectorAll('.clickable-row').forEach(r => r.classList.remove('active-row'));
 document.querySelectorAll('.step-btn').forEach(b => b.classList.remove('active-step'));
 const el = document.getElementById(`m-${m.id}`); if (el) el.classList.add('active-row');
 
 const badge = document.getElementById('hud-label');
 badge.className = "measure-badge pos-bottom-right";
 badge.innerText = m.name;
}

function setStep(stepNum) {
 activeMeasure = null; activeStep = stepNum; activeImgType = "body";
 document.querySelectorAll('.clickable-row').forEach(r => r.classList.remove('active-row'));
 document.querySelectorAll('.step-btn').forEach(b => b.classList.remove('active-step'));
 document.getElementById(`step-${stepNum}`).classList.add('active-step');
 
 const badge = document.getElementById('hud-label');
 badge.className = "measure-badge pos-bottom-center";
 
 const txt = ["Шаг 1: Посох и Мера", "Шаг 2: Учёт Стопы", "Шаг 3: Сопряжение", "Шаг 4: Итог (Ваш Лоб)"];
 badge.innerText = txt[stepNum - 1];
}

function getImgLayout() {
 let img = imgHand; if (activeImgType === "body") img = imgBody; if (activeImgType === "body2") img = imgBody2;
 if (!img.complete || img.naturalWidth === 0) return null;
 const cw = canvas.width, ch = canvas.height, iw = img.naturalWidth, ih = img.naturalHeight;
 const ratio = Math.min(cw / iw, ch / ih);
 return { x: (cw - iw * ratio) / 2, y: (ch - ih * ratio) / 2, width: iw * ratio, height: ih * ratio };
}

function resizeCanvas() {
 const container = document.getElementById('canvas-container');
 if (canvas.width !== container.clientWidth || canvas.height !== container.clientHeight) {
 canvas.width = container.clientWidth; canvas.height = container.clientHeight;
 }
}

canvas.addEventListener('mousedown', function(e) {
 const layout = getImgLayout(); if (!layout) return;
 const rect = canvas.getBoundingClientRect();
 const pctX = ((e.clientX - rect.left - layout.x) / layout.width) * 100;
 const pctY = ((e.clientY - rect.top - layout.y) / layout.height) * 100;
 if (pctX >= 0 && pctX <= 100 && pctY >= 0 && pctY <= 100) {
 document.getElementById('debug-coords').innerText = `x: ${pctX.toFixed(1)}, y: ${pctY.toFixed(1)}`;
 }
});

function drawPulseLine(layout, x1Pct, y1Pct, x2Pct, y2Pct, text) {
 const x1 = layout.x + (x1Pct / 100) * layout.width;
 const y1 = layout.y + (y1Pct / 100) * layout.height;
 const x2 = layout.x + (x2Pct / 100) * layout.width;
 const y2 = layout.y + (y2Pct / 100) * layout.height;
 
 ctx.save();
 ctx.strokeStyle = `rgba(231, 76, 60, ${pulseValue})`;
 ctx.lineWidth = 4; ctx.lineCap = "round";
 ctx.beginPath(); ctx.moveTo(x1, y1); ctx.lineTo(x2, y2); ctx.stroke();
 
 if (text) {
 ctx.fillStyle = "#2c3e50"; ctx.font = "bold 13px sans-serif"; ctx.textAlign = "center";
 ctx.fillText(text, (x1 + x2) / 2, ((y1 + y2) / 2) - 12);
 }
 ctx.restore();
}

function render() {
 resizeCanvas(); ctx.clearRect(0, 0, canvas.width, canvas.height);
 pulseValue += pulseDir; if (pulseValue <= 0.2 || pulseValue >= 1.0) pulseDir = -pulseDir;
 
 const layout = getImgLayout();
 if (layout) {
 let img = imgHand; if (activeImgType === "body") img = imgBody; if (activeImgType === "body2") img = imgBody2;
 ctx.drawImage(img, layout.x, layout.y, layout.width, layout.height);
 
 if (activeMeasure && activeMeasure.lines) {
 activeMeasure.lines.forEach((line, index) => {
 drawPulseLine(layout, line.x1, line.y1, line.x2, line.y2, index === 0 ? activeMeasure.sign : "");
 });
 }
 
 if (activeStep) {
 if (activeStep === 1) {
 const p = measures.medium.find(m => m.id === "posoh");
 const m = measures.medium.find(m => m.id === "mera");
 if (p && p.lines) drawPulseLine(layout, p.lines.x1, p.lines.y1, p.lines.x2, p.lines.y2, "Посох");
 if (m && m.lines) drawPulseLine(layout, m.lines.x1, m.lines.y1, m.lines.x2, m.lines.y2, "Мера");
 }
 if (activeStep === 2) {
 const l = measures.medium.find(m => m.id === "lokot");
 if (l && l.lines) l.lines.forEach(line => drawPulseLine(layout, line.x1, line.y1, line.x2, line.y2, "Локоть канона"));
 }
 if (activeStep === 3) {
 const s = measures.medium.find(m => m.id === "stopa");
 if (s && s.lines) {
 s.lines.forEach(line => drawPulseLine(layout, line.x1, line.y1, line.x2, line.y2, "Стопа"));
 }
 }
 if (activeStep === 4) {
 const lob = measures.medium.find(m => m.id === "lob");
 if (lob && lob.lines) drawPulseLine(layout, lob.lines.x1, lob.lines.y1, lob.lines.x2, lob.lines.y2, "ИТОГ: Ваш Лоб");
 }
 }
 }
 requestAnimationFrame(render);
}

window.onload = () => { buildTables(); setStep(1); render(); };
