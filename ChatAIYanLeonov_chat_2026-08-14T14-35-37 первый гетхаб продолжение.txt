Шаг 2. Код скрипта
Создайте в папке с вашими текстовыми файлами скрипт uploader.py и вставьте этот код:
python
import os
from github import Github
# --- НАСТРОЙКИ ---
TOKEN = "ВАШ_ГИТХАБ_ТОКЕН"         # Вставьте ваш скопированный токен
REPO_NAME = "yanleonov/ai-library-nauka2050"  # Имя вашего репозитория
FILE_TO_UPLOAD = "new_book.txt"    # Имя файла, который хотите отправить
# ------------------
def upload_to_github():
if not os.path.exists(FILE_TO_UPLOAD):
print(f"Ошибка: Файл '{FILE_TO_UPLOAD}' не найден в папке со скриптом!")
return
print("Подключение к GitHub...")
g = Github(TOKEN)
repo = g.get_repo(REPO_NAME)
with open(FILE_TO_UPLOAD, "r", encoding="utf-8") as file:
content = file.read()
# Загружаем файл в корень репозитория (или укажите путь вроде 'texts/' + FILE_TO_UPLOAD)
path_in_repo = FILE_TO_UPLOAD
try:
# Проверяем, существует ли файл, чтобы обновить его
contents = repo.get_contents(path_in_repo)
repo.update_file(contents.path, f"Авто-обновление: {FILE_TO_UPLOAD}", content, contents.sha)
print(f" Успешно обновлено: {FILE_TO_UPLOAD} на GitHub!")
except Exception:
# Если файла нет, создаем новый
repo.create_file(path_in_repo, f"Авто-добавление: {FILE_TO_UPLOAD}", content)
print(f" Успешно загружен новый файл: {FILE_TO_UPLOAD} на GitHub!")
if __name__ == "__main__":
upload_to_github()
Используйте код с осторожностью.
Как пользоваться:
Положите новый текстовый файл в папку со скриптом.
Пропишите имя этого файла в строке FILE_TO_UPLOAD = "имя.txt".
Запустите скрипт через терминал: python uploader.py. Файл автоматически появится в вашем репозитории, и сайт на GitHub Pages обновится самостоятельно!

import os
import re
from github import Github
TOKEN = ""
REPO_NAME = "yanleonov/ai-library-nauka2050"
def auto_upload_txt_files():
g = Github(TOKEN)
repo = g.get_repo(REPO_NAME)
# Ищем все файлы .txt в папке на компьютере
files_in_folder = [f for f in os.listdir('.') if f.endswith('.txt')]
if not files_in_folder:
print("В папке нет .txt файлов для отправки!")
return
print(f"Найдено текстовых файлов на ПК: {len(files_in_folder)}")
# Получаем текущий index.html с GitHub, чтобы обновить ссылки
try:
index_file = repo.get_contents("index.html")
index_html = index_file.decoded_content.decode("utf-8")
except Exception:
print("Ошибка: Не удалось найти index.html на GitHub для авто-обновления ссылок!")
return
is_index_changed = False
for filename in files_in_folder:
with open(filename, "r", encoding="utf-8") as file:
content = file.read()
# Путь к файлу в папке "books" на GitHub
path_in_repo = f"books/{filename}"
# 1. Загружаем или обновляем сам файл книги
try:
contents = repo.get_contents(path_in_repo)
repo.update_file(contents.path, f"Авто-обновление книги: {filename}", content, contents.sha)
print(f" ✓ Обновлен файл книги: {filename}")
except Exception:
repo.create_file(path_in_repo, f"Авто-добавление книги: {filename}", content)
print(f" ✓ Загружен новый файл книги: {filename}")
# 2. Автоматически создаем красивое имя для ссылки на сайте
# Убираем расширение .txt и заменяем нижние подчеркивания на пробелы
clean_name = os.path.splitext(filename)[0].replace('_', ' ')
link_html = f'<li><a href="books/{filename}">{clean_name}</a></li>'
# Если такой ссылки в index.html еще нет, добавляем её внутрь списка <ul>
if f'href="books/{filename}"' not in index_html:
# Ищем закрывающий тег списка </ul> и вставляем новую строку прямо перед ним
index_html = index_html.replace("</ul>", f"  {link_html}\n</ul>")
is_index_changed = True
print(f" + Ссылка на '{clean_name}' добавлена в код index.html")
# 3. Если появились новые ссылки, отправляем обновленный index.html на GitHub
if is_index_changed:
repo.update_file("index.html", "Авто-обновление списка книг в index.html", index_html, index_file.sha)
print(" ✓ Главная страница index.html успешно обновлена на GitHub!")
else:
print(" Все ссылки на сайте уже актуальны, обновление главной страницы не потребовалось.")
if __name__ == "__main__":
auto_upload_txt_files()

Рад, что всё сработало как надо! Команда Ctrl + F5 — это один из самых полезных секретов любого веб-разработчика и создателя сайтов.
Обычное нажатие F5 просто обновляет страницу, но браузер в целях экономии интернета всё равно берёт часть картинок, стилей и старого текста из своей внутренней памяти (кэша). А вот комбинация Ctrl + F5 буквально за шкирку трясёт браузер, заставляя его полностью стереть всё, что он запомнил, и скачать сайт с сервера GitHub абсолютно заново, в его самом свежем виде.
Теперь, когда вы будете обновлять свой сайт после добавления новых книг через наш кликер, используйте именно Ctrl + F5 — тогда все новые ссылки будут появляться на экране мгновенно, без задержек.
