    function generateGeometry(scaleType, isTriadForm) {
        currentPoints = [];
        let step = 30; 
        
        let ballCount = 8;
        let isCosmicModel = false;
        
        if (scaleType === "micro") ballCount = 4;
        else if (scaleType === "small") ballCount = 6;
        else if (scaleType === "medium") ballCount = 12;
        else if (scaleType === "large") ballCount = 20;
        else if (scaleType === "cosmic") { ballCount = 30; isCosmicModel = true; }

        if (!isTriadForm) {
            if (!isCosmicModel) {
                for (let i = 0; i <= ballCount; i++) {
                    currentPoints.push({ x: i * step, y: 0, z: 0, isNode: i === 0 || i === ballCount, isGold: i === 0 || i === ballCount });
                    if (i > 0 && i < ballCount && ballCount < 15) {
                        currentPoints.push({ x: i * step, y: step * 0.35, z: 0, isNode: false, isGold: false });
                        currentPoints.push({ x: i * step, y: -step * 0.35, z: 0, isNode: false, isGold: false });
                    }
                }
            } else {
                for (let i = 0; i < ballCount; i++) {
                    let angle = (i * 2 * Math.PI) / ballCount;
                    let radius = step * 3;
                    currentPoints.push({ x: Math.cos(angle) * radius, y: 0, z: Math.sin(angle) * radius, isNode: i === 0, isGold: i === 0 });
                }
                currentPoints.push({ x: 0, y: 0, z: 0, isNode: true, isGold: true, isSun: true });
            }
        } else {
            let hStep = step * Math.sqrt(2 / 3);
            currentPoints.push({ x: 0, y: hStep, z: 0, isNode: true, isGold: true });
            let r = step / Math.sqrt(3);
            for (let a = 0; a < 3; a++) {
                let angle = (a * 2 * Math.PI) / 3;
                currentPoints.push({ x: Math.cos(angle) * r, y: 0, z: Math.sin(angle) * r, isNode: true, isGold: true });
            }
            currentPoints.push({ x: 0, y: -hStep, z: 0, isNode: true, isGold: true });
        }

        if (currentPoints.length > 0) {
            let minX = currentPoints[0].x, maxX = currentPoints[0].x;
            let minY = currentPoints[0].y, maxY = currentPoints[0].y;
            let minZ = currentPoints[0].z, maxZ = currentPoints[0].z;
            
            currentPoints.forEach(p => {
                if (p.x < minX) minX = p.x; if (p.x > maxX) maxX = p.x;
                if (p.y < minY) minY = p.y; if (p.y > maxY) maxY = p.y;
                if (p.z < minZ) minZ = p.z; if (p.z > maxZ) maxZ = p.z;
            });
            
            let midX = (minX + maxX) / 2;
            let midY = (minY + maxY) / 2;
            let midZ = (minZ + maxZ) / 2;

            currentPoints.forEach(p => {
                p.x = p.x - midX;
                p.y = p.y - midY;
                p.z = p.z - midZ;
            });
        }

let maxDist = 1;

function renderScene() {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        let cx = canvas.width / 2, cy = canvas.height / 2;

        let cosX = Math.cos(angleX), sinX = Math.sin(angleX);
        let cosY = Math.cos(angleY), sinY = Math.sin(angleY);

        let projected = currentPoints.map(p => {
            let x1 = p.x * cosY - p.z * sinY;
            let z1 = p.x * sinY + p.z * cosY;
            let y2 = p.y * cosX - z1 * sinX;
            let z2 = p.y * sinX + z1 * cosX;
            return { x: cx + x1 * scale3D, y: cy - y2 * scale3D, z: z2, isGold: p.isGold, isNode: p.isNode, isSun: p.isSun };
        });

            projected.sort((a, b) => a.z - b.z);

            projected.forEach(p => {
                    let radius = (p.isNode ? 13 : 8) * scale3D;
                    if (p.isSun) radius = 22 * scale3D;
            
                    let grad = ctx.createRadialGradient(p.x - radius * 0.1, p.y - radius * 0.1, 1, p.x, p.y, radius);
            ctx.shadowBlur = 0; 
            
            if (p.isSun) {
                grad.addColorStop(0, '#ffff00'); grad.addColorStop(0.4, '#ff3d00'); grad.addColorStop(1, '#dd2c00');
            } else if (p.isGold) {
                grad.addColorStop(0, '#fff3cc'); grad.addColorStop(0.3, '#ffbb00'); grad.addColorStop(1, '#805500');
            } else {
                grad.addColorStop(0, '#e8f5e9'); grad.addColorStop(0.3, '#4caf50'); grad.addColorStop(1, '#1b5e20');
            }
            
            ctx.beginPath(); ctx.arc(p.x, p.y, radius, 0, Math.PI * 2);
            ctx.fillStyle = grad; ctx.fill(); ctx.closePath();
        });
    }

    function setupMouseEvents() {
        canvas.addEventListener('mousedown', (e) => {
            isDragging = true;
            previousMousePosition = { x: e.clientX, y: e.clientY };
        });
        window.addEventListener('mousemove', (e) => {
            if (!isDragging) return;
            let deltaX = e.clientX - previousMousePosition.x;
            let deltaY = e.clientY - previousMousePosition.y;
            angleY += deltaX * 0.01;
            angleX += deltaY * 0.01; 
            previousMousePosition = { x: e.clientX, y: e.clientY };
            renderScene();
        });
        window.addEventListener('mouseup', () => isDragging = false);
        canvas.addEventListener('wheel', (e) => {
            e.preventDefault();
            scale3D *= e.deltaY < 0 ? 1.1 : 0.9;
            renderScene();
        }, { passive: false });
    }

    function onWindowResize() {
        const visualizer = canvas.parentElement;
        canvas.width = visualizer.clientWidth || 650;
        canvas.height = visualizer.clientHeight || 460;
        if(currentPoints.length > 0) {
            updateSelection(selectedRowIndex);
        }
    }
