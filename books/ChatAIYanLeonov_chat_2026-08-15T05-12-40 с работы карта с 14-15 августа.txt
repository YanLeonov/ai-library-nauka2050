import os
from qgis.core import QgsVectorLayer, QgsProject, QgsPalLayerSettings, QgsTextFormat, QgsVectorLayerSimpleLabeling, QgsTextBufferSettings
# Наш точный путь к папке
folder = "F:/Рабочий стол/Ялышев К А/Изображения/Карта Нижневартовска/ural-fed-district-260813-free.shp"
# Границы Нижневартовска для фильтрации
bbox_expression = "within(geom_from_wkt('POLYGON((76.51 60.91, 76.63 60.91, 76.63 60.97, 76.51 60.97, 76.51 60.91))'))"
print("Сканируем все файлы архива на наличие номеров и улиц...")
# Настройки стиля для текста
def apply_big_labels(layer, field_name, size):
settings = QgsPalLayerSettings()
settings.fieldName = field_name
text_format = QgsTextFormat()
text_format.setSize(size)
buf = QgsTextBufferSettings()
buf.setEnabled(True)
buf.setSize(2.0)
text_format.setBuffer(buf)
settings.setFormat(text_format)
layer.setLabeling(QgsVectorLayerSimpleLabeling(settings))
layer.setLabelsEnabled(True)
# Перебираем все файлы в папке
for file in os.listdir(folder):
if file.endswith(".shp"):
path = os.path.join(folder, file)
layer_name = file.replace(".shp", "")
layer = QgsVectorLayer(path, layer_name, "ogr")
if layer.isValid():
fields = [f.name().lower() for f in layer.fields()]
# 1. Ищем номера домов (они могут быть в housenumber, name или addr_house)
if 'housenumber' in fields:
layer.setSubsetString(bbox_expression)
if layer.featureCount() > 0:
layer.setName(f"Номера домов ({layer_name})")
QgsProject.instance().addMapLayer(layer)
apply_big_labels(layer, layer.fields().at(fields.index('housenumber')).name(), 24)
print(f"-> Найдено и добавлено поле номеров в слое: {file}")
# 2. Ищем названия улиц (только в линейных слоях дорог, где fclass = 'highway' или имя слоя содержит 'roads')
elif 'name' in fields and 'roads' in layer_name.lower():
layer.setSubsetString(bbox_expression)
if layer.featureCount() > 0:
layer.setName(f"Названия улиц ({layer_name})")
QgsProject.instance().addMapLayer(layer)
apply_big_labels(layer, 'name', 20)
print(f"-> Найдено и добавлено поле улиц в слое: {file}")
print("Поиск и настройка завершены!")
