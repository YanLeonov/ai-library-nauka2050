<script>
    let canvas, ctx;
    let selectedRowIndex = 0;
    
    let angleX = -0.4, angleY = 0.5; 
    let scale3D = 1.0;
    let isDragging = false;
    let previousMousePosition = { x: 0, y: 0 };
    let currentPoints = [];

    // Четкий расчет числового ряда по книге: от n=2 до n=10 (ровно 9 строк)
    const pyramidValues = [];
    const mappingN = []; 
    let currentSum = 1;  // n=1 равен 1, в таблицу не идет
    
    for (let n = 2; n <= 10; n++) {
        currentSum += n * n;
        pyramidValues.push(currentSum);
        mappingN.push(n);
    }

    function init() {
        canvas = document.getElementById('canvas3d');
        ctx = canvas.getContext('2d');
        buildTableAutomated();
        setupMouseEvents();
        updateSelection(0);
        window.addEventListener('keydown', handleKeyboard);
        window.addEventListener('resize', onWindowResize);
        onWindowResize();
    }

    function buildTableAutomated() {
        const tbody = document.getElementById('table-body');
        tbody.innerHTML = '';
        for (let i = 0; i < 9; i++) {
            const n = mappingN[i];
            const valPyramid = pyramidValues[i];
            
            const tr = document.createElement('tr');
            tr.id = `row-${i}`;
            
            const tdPyramid = document.createElement('td');
            tdPyramid.id = `cell-pyramid-${i}`;
            tdPyramid.className = "clickable-cell";
            tdPyramid.innerHTML = `П × ${n} ≡ П<sup>${n}</sup> = ${valPyramid}`;
            tdPyramid.onclick = function() { updateSelection(i); };

            tr.appendChild(tdPyramid); tbody.appendChild(tr);
        }
    }

    function updateSelection(index) {
        selectedRowIndex = index;
        document.querySelectorAll('.clickable-cell').forEach(cell => cell.classList.remove('selected-item'));
        const targetCellId = `cell-pyramid-${index}`;
        const targetCell = document.getElementById(targetCellId);
        if(targetCell) {
            targetCell.classList.add('selected-item');
            targetCell.scrollIntoView({ block: 'nearest', behavior: 'smooth' });
        }
        generateGeometry(mappingN[index]);
        renderScene();
    }

    function handleKeyboard(e) {
        if (e.key === "ArrowDown" && selectedRowIndex < 8) { e.preventDefault(); updateSelection(selectedRowIndex + 1); }
        if (e.key === "ArrowUp" && selectedRowIndex > 0) { e.preventDefault(); updateSelection(selectedRowIndex - 1); }
    }
    // Геометрия четырехугольной пирамиды на базе квадратных Ровн
    function generateGeometry(rows) {
        currentPoints = [];
        let step = 30; 
        
        // Строим слои сверху вниз (от h=0 - одиночная вершина, до h=rows-1 - основание)
        for (let h = 0; h < rows; h++) {
            let sideCount = h + 1;
            // Физический шаг высоты четырехугольной пирамиды из шаров
            let yPos = h * step * (1 / Math.sqrt(2));
            
            // Центрируем каждый квадратный слой относительно абсолютного начала координат
            let startOffset = -(sideCount - 1) * step * 0.5;
            
            for (let r = 0; r < sideCount; r++) {
                for (let c = 0; c < sideCount; c++) {
                    currentPoints.push({
                        x: startOffset + c * step,
                        y: yPos,
                        z: startOffset + r * step
                    });
                }
            }
        }

        // Автоматическое точное центрирование всей собранной фигуры по высоте Y
        let minY = currentPoints[0].y;
        let maxY = currentPoints[0].y;
        currentPoints.forEach(p => {
            if (p.y < minY) minY = p.y;
            if (p.y > maxY) maxY = p.y;
        });
        let midY = (minY + maxY) / 2;

        currentPoints.forEach(p => {
            p.y = -(p.y - midY); // Переворачиваем вершиной вверх и центрируем по оси Y
        });

        // Жесткий расчет maxDist для предотвращения ошибок деления
        let maxDist = 1;
        currentPoints.forEach(p => {
            let d = Math.sqrt(p.x*p.x + p.y*p.y + p.z*p.z);
            if(d > maxDist) maxDist = d;
        });
        
        scale3D = ((canvas.width * 0.32) / maxDist) * 0.65;
    }

    function renderScene() {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        let cx = canvas.width / 2, cy = canvas.height / 2;

        let cosX = Math.cos(angleX), sinX = Math.sin(angleX);
        let cosY = Math.cos(angleY), sinY = Math.sin(angleY);

        let projected = currentPoints.map(p => {
            let x1 = p.x * cosY - p.z * sinY;
            let z1 = p.x * sinY + p.z * cosY;
            
            let y2 = p.y * cosX - z1 * sinX;
            let z2 = p.y * sinX + z1 * cosX;
            
            return { x: cx + x1 * scale3D, y: cy - y2 * scale3D, z: z2 };
        });

        // Честный Z-буфер сортировки
        projected.sort((a, b) => a.z - b.z);

        projected.forEach(p => {
            let radius = 15 * scale3D;
            let grad = ctx.createRadialGradient(p.x - radius * 0.1, p.y - radius * 0.1, 1, p.x, p.y, radius);
            ctx.shadowBlur = 0; 
            
            // Комфортная тёмно-матовая бордовая палитра
            grad.addColorStop(0, '#ff9e80'); 
            grad.addColorStop(0.3, '#ff3d00'); 
            grad.addColorStop(1, '#9a0000');
            
            ctx.beginPath(); ctx.arc(p.x, p.y, radius, 0, Math.PI * 2);
            ctx.fillStyle = grad; ctx.fill(); ctx.closePath();
        });
    }

    function setupMouseEvents() {
        canvas.addEventListener('mousedown', (e) => {
            isDragging = true;
            previousMousePosition = { x: e.clientX, y: e.clientY };
        });
        window.addEventListener('mousemove', (e) => {
            if (!isDragging) return;
            let deltaX = e.clientX - previousMousePosition.x;
            let deltaY = e.clientY - previousMousePosition.y;
            
            angleY += deltaX * 0.01;
            angleX += deltaY * 0.01; 
            
            previousMousePosition = { x: e.clientX, y: e.clientY };
            renderScene();
        });
        window.addEventListener('mouseup', () => isDragging = false);
        canvas.addEventListener('wheel', (e) => {
            e.preventDefault();
            scale3D *= e.deltaY < 0 ? 1.1 : 0.9;
            renderScene();
        }, { passive: false });
    }

    function onWindowResize() {
        const visualizer = canvas.parentElement;
        canvas.width = visualizer.clientWidth || 650;
        canvas.height = visualizer.clientHeight || 460;
        if(currentPoints.length > 0) {
            updateSelection(selectedRowIndex);
        }
    }
    window.onload = init;
</script>
