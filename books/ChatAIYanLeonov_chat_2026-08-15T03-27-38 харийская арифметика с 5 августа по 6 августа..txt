function generateAllData() {
const data = {
'Гармоническая': { title: 'Гармоническая система («два на...» и «два жды...»)', rows: [] },
'Триадная': { title: 'Триадная система («на...» и «жды...»)', rows: [] },
'Ровная': { title: 'Ровная система («ровно на...» и «ровно жды...»)', rows: [] },
'Пирамидальная': { title: 'Система Пирамидального Умножения («пирамида жды...»)', rows: [] },
'Призменная': { title: 'Призменная система (Малая и Ровная призма жды...)', rows: [] }
};
// Генерируем ряды от 2 до 65 строго по таблицам из книги
for (let n = 2; n <= 65; n++) {
// 1. ГАРМОНИЧЕСКАЯ
data['Гармоническая'].rows.push([`2 &middot; ${n}`, `2<sub>-1</sub><sup>${n}</sup>`, `${2 * n}`]);
data['Гармоническая'].rows.push([`2 &times; ${n}`, `2<sub>-1</sub><sup>${n}</sup> (куб)`, `${2 * n * 4}`]);
// 2. ТРИАДНАЯ (Формулы треугольных и тетраэдрических чисел)
let tri2d = (n * (n + 1)) / 2;
let tri3d = (n * (n + 1) * (n + 2)) / 6;
data['Триадная'].rows.push([`3<sub>-1</sub> &middot; ${n}`, `3<sub>-1</sub><sup>${n}</sup>`, `${tri2d}`]);
data['Триадная'].rows.push([`3<sub>-1</sub> &times; ${n}`, `3<sub>-1</sub><sup>${n}</sup> (ЖДЫ)`, `${tri3d}`]);
// 3. РОВНАЯ (Квадраты и Кубы)
data['Ровная'].rows.push([`4<sub>-1</sub> &middot; ${n}`, `4<sub>-1</sub><sup>${n}</sup> (Плоская)`, `${n * n}`]);
data['Ровная'].rows.push([`4<sub>-1</sub> &times; ${n}`, `4<sub>-1</sub><sup>${n}</sup> (Объемная)`, `${n * n * n}`]);
// 4. ПИРАМИДАЛЬНАЯ (Сумма квадратов — Пирамида жды...)
let pyr = (n * (n + 1) * (2 * n + 1)) / 6;
data['Пирамидальная'].rows.push([`&Pi; &times; ${n}`, `&Pi;<sup>${n}</sup>`, `${pyr}`]);
}
// 5. ПРИЗМЕННАЯ (Малая П и Ровная R призма до 16 из листов 11, 13)
for (let n = 2; n <= 16; n++) {
let p_rows = n * 2 - 1;
let p_res = (n * (n + 1) * (2 * n + 1)) / 6; // Малая призма равна пирамиде по точкам
data['Призменная'].rows.push([`P &times; ${p_rows}`, `P<sup>${n}</sup> (Малая, ${n} основ)`, `${p_res}`]);
}
for (let n = 2; n <= 16; n++) {
let r_rows = n * 2 - 1;
let r_res = ((n * (n + 1) * (2 * n + 1)) / 6) + (((n-1) * n * (2 * (n-1) + 1)) / 6) + (n * n); // Ровная призма
data['Призменная'].rows.push([`R &times; ${r_rows}`, `R<sup>${n}</sup> (Ровная, ${n} основ)`, `${r_res}`]);
}
return data;
}
