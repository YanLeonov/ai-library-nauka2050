<script>
    let canvas, ctx;
    let selectedRowIndex = 0, isCurrentVolume = false;
    
    let angleX = -0.5, angleY = 0.6;
    let scale3D = 1.0;
    let isDragging = false;
    let previousMousePosition = { x: 0, y: 0 };
    let currentPoints = [];

    function init() {
        canvas = document.getElementById('canvas3d');
        ctx = canvas.getContext('2d');
        buildTableAutomated();
        setupMouseEvents();
        updateSelection(0, false);
        window.addEventListener('keydown', handleKeyboard);
        window.addEventListener('resize', onWindowResize);
        onWindowResize();
    }

    function buildTableAutomated() {
        const tbody = document.getElementById('table-body');
        tbody.innerHTML = '';
        for (let i = 0; i <= 63; i++) {
            let n = i + 2;
            const tr = document.createElement('tr');
            tr.id = `row-${i}`;
            const valNa = 2 * n, valZhdy = 8 * n;
            
            const tdNa = document.createElement('td');
            tdNa.id = `cell-na-${i}`;
            tdNa.className = "clickable-cell";
            tdNa.innerText = `2 · ${n} = ${valNa}`;
            tdNa.onclick = function() { updateSelection(i, false); };

            const tdZhdy = document.createElement('td');
            tdZhdy.id = `cell-zhdy-${i}`;
            tdZhdy.className = "clickable-cell";
            tdZhdy.innerText = `2 × ${n} = ${valZhdy}`;
            tdZhdy.onclick = function() { updateSelection(i, true); };

            tr.appendChild(tdNa); tr.appendChild(tdZhdy); tbody.appendChild(tr);
        }
    }

    function updateSelection(index, isVolume) {
        selectedRowIndex = index; isCurrentVolume = isVolume;
        document.querySelectorAll('.clickable-cell').forEach(cell => cell.classList.remove('selected-item'));
        const targetCellId = isVolume ? `cell-zhdy-${index}` : `cell-na-${index}`;
        const targetCell = document.getElementById(targetCellId);
        if(targetCell) {
            targetCell.classList.add('selected-item');
            targetCell.scrollIntoView({ block: 'nearest', behavior: 'smooth' });
        }
        generateGeometry(isVolume ? (8 * (index + 2)) : (2 * (index + 2)), isVolume);
        renderScene();
    }

    function handleKeyboard(e) {
        if (e.key === "ArrowDown" && selectedRowIndex < 63) { e.preventDefault(); updateSelection(selectedRowIndex + 1, isCurrentVolume); }
        if (e.key === "ArrowUp" && selectedRowIndex > 0) { e.preventDefault(); updateSelection(selectedRowIndex - 1, isCurrentVolume); }
    }

    function generateGeometry(count, isVolume) {
        currentPoints = [];
        if (!isVolume) {
            let cols = Math.ceil(Math.sqrt(count));
            // Возвращаем ИСХОДНЫЙ шаг 40 из самой первой правильной версии
            let step = 40; 
            for (let i = 0; i < count; i++) {
                let r = i % cols;
                let c = Math.floor(i / cols);
                currentPoints.push({
                    x: (r - (cols-1)/2) * step,
                    y: 0,
                    z: (c - (Math.ceil(count/cols)-1)/2) * step,
                    isGold: false
                });
            }
        } else {
            let cubeGroups = count / 8, stepX = 40; 
            for (let g = 0; g < cubeGroups; g++) {
                let offsetX = (g - (cubeGroups - 1) / 2) * stepX;
                // Восстанавливаем оригинальные пропорции кубических блоков
                currentPoints.push({ x: -10 + offsetX, y: -10, z: -10, isGold: true });
                currentPoints.push({ x:  10 + offsetX, y: -10, z: -10, isGold: true });
                currentPoints.push({ x: -10 + offsetX, y:  10, z: -10, isGold: true });
                currentPoints.push({ x:  10 + offsetX, y:  10, z: -10, isGold: true });
                currentPoints.push({ x: -10 + offsetX, y: -10, z:  10, isGold: true });
                currentPoints.push({ x:  10 + offsetX, y: -10, z:  10, isGold: true });
                currentPoints.push({ x: -10 + offsetX, y:  10, z:  10, isGold: true });
                currentPoints.push({ x:  10 + offsetX, y:  10, z:  10, isGold: true });
            }
        }
        let maxDist = 1;
        currentPoints.forEach(p => {
            let d = Math.sqrt(p.x*p.x + p.y*p.y + p.z*p.z);
            if(d > maxDist) maxDist = d;
        });
        // Ограничиваем базовый масштаб сцены, чтобы фигуры гарантированно влезали в окно
        scale3D = (canvas.width * 0.35) / maxDist;
    }

    function renderScene() {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        let cx = canvas.width / 2, cy = canvas.height / 2;

        let cosX = Math.cos(angleX), sinX = Math.sin(angleX);
        let cosY = Math.cos(angleY), sinY = Math.sin(angleY);

        let projected = currentPoints.map(p => {
            let x1 = p.x * cosY - p.z * sinY;
            let z1 = p.x * sinY + p.z * cosY;
            let y2 = p.y * cosX - z1 * sinX;
            let z2 = p.y * sinX + z1 * cosX;
            return { x: cx + x1 * scale3D, y: cy - y2 * scale3D, z: z2, isGold: p.isGold };
        });

        projected.sort((a, b) => b.z - a.z);

        projected.forEach(p => {
            // ОБЩАЯ ЗАВИСИМОСТЬ: Возвращена связь радиуса с динамическим масштабом scale3D, 
            // но коэффициент уменьшен строго в два раза (до 0.5), чтобы шары не раздувались
            let radius = (p.isGold ? 10 : 20) * scale3D * 0.5; 
            if (radius < 1) radius = 1; 
            
            let grad = ctx.createRadialGradient(p.x - radius*0.3, p.y - radius*0.3, 1, p.x, p.y, radius);
            ctx.shadowBlur = 4;
            if (p.isGold) {
                ctx.shadowColor = "rgba(255, 170, 0, 0.3)";
                grad.addColorStop(0, '#fff3cc'); grad.addColorStop(0.3, '#ffbb00'); grad.addColorStop(1, '#805500');
            } else {
                ctx.shadowColor = "rgba(0, 168, 255, 0.3)";
                grad.addColorStop(0, '#e6f7ff'); grad.addColorStop(0.3, '#00a8ff'); grad.addColorStop(1, '#004466');
            }
            ctx.beginPath(); ctx.arc(p.x, p.y, radius, 0, Math.PI * 2);
            ctx.fillStyle = grad; ctx.fill(); ctx.closePath();
        });
    }

    function setupMouseEvents() {
        canvas.addEventListener('mousedown', (e) => {
            isDragging = true;
            previousMousePosition = { x: e.clientX, y: e.clientY };
        });
        window.addEventListener('mousemove', (e) => {
            if (!isDragging) return;
            let deltaX = e.clientX - previousMousePosition.x;
            let deltaY = e.clientY - previousMousePosition.y;
            angleY -= deltaX * 0.01;
            angleX -= deltaY * 0.01;
            previousMousePosition = { x: e.clientX, y: e.clientY };
            renderScene();
        });
        window.addEventListener('mouseup', () => isDragging = false);
        canvas.addEventListener('wheel', (e) => {
            e.preventDefault();
            scale3D *= e.deltaY < 0 ? 1.1 : 0.9;
            renderScene();
        }, { passive: false });
    }

    function onWindowResize() {
        const visualizer = canvas.parentElement;
        canvas.width = visualizer.clientWidth || 650;
        canvas.height = visualizer.clientHeight || 460;
        if(currentPoints.length > 0) {
            updateSelection(selectedRowIndex, isCurrentVolume);
        }
    }
    window.onload = init;
</script>
