        <div class="canvas-title" id="display-title">Плоскостное умножение "ЖДЫ"</div>
        <canvas id="localCanvas" width="400" height="400"></canvas>
        <div class="canvas-footer" id="display-footer">Раскрытие одной меры в плоскость 3х3 (9 структур)</div>

    <script>
        const canvas = document.getElementById('localCanvas');
        const ctx = canvas.getContext('2d');
        let isDrawing = false; 
        let currentMode = 'single'; 
        let userInteractionMode = 'light'; 
        
        let offsetX = 0; let offsetY = 0;
        let angleX = 0;  let angleY = 0;

        const flatGrid9Balls = [];
        const stepFlat = 54; 
        for (let x = -1; x <= 1; x++) {
            for (let y = -1; y <= 1; y++) {
                flatGrid9Balls.push({ x: x * stepFlat, y: y * stepFlat, z: 0, r: 28 });
            }
        }

        const initialBalls3D = {
            single: [{ x: 0, y: 0, z: 0, r: 120 }],
            flat_grid9: flatGrid9Balls
        };

        let screenBalls = [];

        function drawSingleBall(cX, cY, radius) {
            let lightX = cX + offsetX;
            let lightY = cY + offsetY;

            let gradient = ctx.createRadialGradient(lightX, lightY, 2, cX, cY, radius);
            gradient.addColorStop(0, '#ffffff');   
            gradient.addColorStop(0.3, '#f87171'); 
            gradient.addColorStop(0.8, '#991b1b'); 
            gradient.addColorStop(1, '#450a0a');   
            
            ctx.beginPath();
            ctx.arc(cX, cY, radius, 0, Math.PI * 2);
            ctx.fillStyle = gradient;
            ctx.fill();
        }

        function renderScene() {
            ctx.clearRect(0, 0, 400, 400);
            if (!isDrawing) return;

            const baseBalls = initialBalls3D[currentMode];
            screenBalls = [];

            baseBalls.forEach(ball => {
                let x1 = ball.x * Math.cos(angleX) - ball.z * Math.sin(angleX);
                let z1 = ball.x * Math.sin(angleX) + ball.z * Math.cos(angleX);
                
                let y2 = ball.y * Math.cos(angleY) - z1 * Math.sin(angleY);
                let z2 = ball.y * Math.sin(angleY) + z1 * Math.cos(angleY);

                screenBalls.push({ x: x1 + 200, y: y2 + 200, z: z2, r: ball.r });
            });

            screenBalls.sort((a, b) => a.z - b.z);
            screenBalls.forEach(ball => { drawSingleBall(ball.x, ball.y, ball.r); });
        }

        function isInsideAnyBall(mouseX, mouseY) {
            for (let ball of screenBalls) {
                const dx = mouseX - ball.x;
                const dy = mouseY - ball.y;
                if (Math.sqrt(dx * dx + dy * dy) <= ball.r) return true;
            }
            return false;
        }

        const table = document.getElementById('math-table');
        const mainHeader = document.getElementById('main-header');
        const displayTitle = document.getElementById('display-title');
        const displayFooter = document.getElementById('display-footer');
        const btnLight = document.getElementById('mode-light');
        const btnFigure = document.getElementById('mode-figure');

        function updatePageStyle(isSlavic) {
            if (isSlavic) {
                table.classList.add('ancient-slavic-style');
                mainHeader.classList.add('ancient-slavic-style');
                displayTitle.classList.add('ancient-slavic-style');
                displayFooter.classList.add('ancient-slavic-style');
                btnLight.classList.add('ancient-slavic-style');
                btnFigure.classList.add('ancient-slavic-style');
                
                mainHeader.innerText = 'Х`Арийская арифметика';
                btnLight.innerText = 'Плоскостной видъ';
                btnFigure.innerText = 'Видъ въ объёмѣ';

                if (currentMode === 'single' || currentMode === 'flat_grid9') {
                    displayTitle.innerText = 'Плоскостное умноженіе "ЖДЫ"';
                    displayFooter.innerText = 'Раскрытіе одной мѣры въ плоскость 3х3 (9 структуръ)';
                } else {
                    displayTitle.innerText = 'Объёмное умноженіе "Ю"';
                    displayFooter.innerText = 'Раскрытіе мѣрности во всѣ стороны пространства';
                }
            } else {
                table.classList.remove('ancient-slavic-style');
                mainHeader.classList.remove('ancient-slavic-style');
                displayTitle.classList.remove('ancient-slavic-style');
                displayFooter.classList.remove('ancient-slavic-style');
                btnLight.classList.remove('ancient-slavic-style');
                btnFigure.classList.remove('ancient-slavic-style');
                
                mainHeader.innerText = 'Х`Арийская арифметика';
                btnLight.innerText = 'Плоскостной вид';
                btnFigure.innerText = 'Вид в объеме';

                if (currentMode === 'single' || currentMode === 'flat_grid9') {
                    displayTitle.innerText = 'Плоскостное умножение "ЖДЫ"';
                    displayFooter.innerText = 'Раскрытие одной меры в плоскость 3х3 (9 структур)';
                } else {
                    displayTitle.innerText = 'Объёмное умножение "Ю"';
                    displayFooter.innerText = 'Раскрытие мерности во все стороны пространства';
                }
            }
        }

        document.getElementById('modern-font').addEventListener('change', () => {
            updatePageStyle(false);
        });
        document.getElementById('slavic-font').addEventListener('change', () => {
            updatePageStyle(true);
        });

        btnLight.addEventListener('click', () => {
            userInteractionMode = 'light';
            btnLight.classList.add('active');
            btnFigure.classList.remove('active');
            angleX = 0; angleY = 0;
            renderScene();
        });
        
        btnFigure.addEventListener('click', () => {
            userInteractionMode = 'figure';
            btnFigure.classList.add('active');
            btnLight.classList.remove('active');
        });

        document.querySelectorAll('.ball-cell').forEach(cell => {
            cell.addEventListener('click', (e) => {
                isDrawing = true;
                currentMode = e.target.getAttribute('data-type');
                angleX = 0; angleY = 0;
                offsetX = 0; offsetY = 0;
                
                const isSlavicActive = document.getElementById('slavic-font').checked;
                if (currentMode === 'single' || currentMode === 'flat_grid9') {
                    document.getElementById('math-sign').innerText = '×';
                    displayTitle.innerText = isSlavicActive ? 'Плоскостное умноженіе "ЖДЫ"' : 'Плоскостное умножение "ЖДЫ"';
                    displayFooter.innerText = isSlavicActive ? 'Раскрытіе одной мѣры въ плоскость 3х3 (9 структуръ)' : 'Раскрытие одной меры в плоскость 3х3 (9 структур)';
                } else {
                    document.getElementById('math-sign').innerText = '·';
                    displayTitle.innerText = isSlavicActive ? 'Объёмное умноженіе "Ю"' : 'Объёмное умножение "Ю"';
                    displayFooter.innerText = isSlavicActive ? 'Раскрытіе мѣрности во всѣ стороны пространства' : 'Раскрытие мерности во все стороны пространства';
                }
                
                renderScene();
            });
        });

        canvas.addEventListener('mousedown', (e) => {
            if (!isDrawing) return;
            
            const rect = canvas.getBoundingClientRect();
            const clickX = e.clientX - rect.left;
            const clickY = e.clientY - rect.top;

            if (!isInsideAnyBall(clickX, clickY)) return;

            canvas.style.cursor = 'grabbing';
            let startX = e.clientX;
            let startY = e.clientY;
            
            let baseOffsetX = offsetX;
            let baseOffsetY = offsetY;
            let baseAngleX = angleX;
            let baseAngleY = angleY;
            
            function move(me) {
                const currentX = me.clientX - rect.left;
                const currentY = me.clientY - rect.top;

                if (!isInsideAnyBall(currentX, currentY)) {
                    up();
                    return;
                }

                let activeInteractionMode = (currentMode === 'single') ? 'light' : userInteractionMode;

                if (activeInteractionMode === 'light') {
                    let targetOffsetX = baseOffsetX + (me.clientX - startX) * 0.5;
                    let targetOffsetY = baseOffsetY + (me.clientY - startY) * 0.5;
                    
                    let distance = Math.sqrt(targetOffsetX * targetOffsetX + targetOffsetY * targetOffsetY);
                    let maxRadiusLimit = (currentMode === 'single') ? 95 : 16; 
                    
                    if (distance <= maxRadiusLimit) {
                        offsetX = targetOffsetX;
                        offsetY = targetOffsetY;
                    } else {
                        offsetX = (targetOffsetX / distance) * maxRadiusLimit;
                        offsetY = (targetOffsetY / distance) * maxRadiusLimit;
                    }
                } else {
                    angleX = baseAngleX - (me.clientX - startX) * 0.015;
                    angleY = baseAngleY - (me.clientY - startY) * 0.015;
                }
                renderScene();
            }
            
            function up() {
                canvas.style.cursor = 'default';
                window.removeEventListener('mousemove', move);
                window.removeEventListener('mouseup', up);
            }
            
            window.addEventListener('mousemove', move);
            window.addEventListener('mouseup', up);
        });

        ctx.clearRect(0, 0, 400, 400);
    </script>
