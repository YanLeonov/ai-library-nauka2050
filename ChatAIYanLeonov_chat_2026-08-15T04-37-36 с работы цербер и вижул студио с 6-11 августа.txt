xml
<Project Sdk="Microsoft.NET.Sdk">
<PropertyGroup>
<OutputType>Exe</OutputType>
<TargetFrameworks>net8.0-android</TargetFrameworks>
<SupportedOSPlatformVersion>21.0</SupportedOSPlatformVersion>
<ApplicationId>com.companyname.cerberus</ApplicationId>
<ApplicationVersion>1</ApplicationVersion>
<ApplicationDisplayVersion>1.0</ApplicationDisplayVersion>
<AndroidPackageFormat>apk</AndroidPackageFormat>
<AndroidBuildApplicationPackage>true</AndroidBuildApplicationPackage>
</PropertyGroup>
<ItemGroup>
<PackageReference Include="Avalonia.Android" Version="11.0.10" />
<PackageReference Include="Avalonia.ReactiveUI" Version="11.0.10" />
</ItemGroup>
<ItemGroup>
<ProjectReference Include="..\Cerberus.Shared\Cerberus.Shared.csproj" />
</ItemGroup>
</Project>

Для прямой связи телефона Samsung M12 (через разъем Type-C) с прибором «Цербер GP-4» (по двухпроводной шине RS-485: линии A и B) вам обязательно нужен физический адаптер-конвертер USB-to-RS485 на базе чипа CH340, CP2102 или FTDI, соединенный через OTG-кабель. Напрямую скрутить провода Type-C с линиями A и B нельзя — сгорят порты, так как в Type-C идут логические уровни 0-5В, а в RS-485 — дифференциальный сигнал до 12В.
Схема распайки проводов от телефона к прибору выглядит следующим образом:
Телефон Samsung M12 ➡️ Провод Type-C (с поддержкой OTG) ➡️ Вход конвертера USB-to-RS485
Выход конвертера клемма 'A' (или D+) ➡️ Провод связи ➡️ Клемма 'A' прибора «Цербер»
Выход конвертера клемма 'B' (или D-) ➡️ Провод связи ➡️ Клемма 'B' прибора «Цербер»
Чтобы Android-приложение, которое мы собираем, смогло слать команды через этот адаптер, стандартный код Windows (работающий с System.IO.Ports) нужно заменить на Android-драйвер.
Ниже представлен готовый класс для Android-версии программы «Цербер», написанный целиком. Он автоматически находит подключенный к Type-C конвертер RS-485 (на любом популярном чипе CH340/CP210X/FTDI) и открывает канал связи.

using System;
using System.Collections.Generic;
using System.IO;
using Android.Content;
using Android.Hardware.Usb;
using Ho顺.UsbSerial.Driver; // Требуется NuGet-пакет usb-serial-for-android в MAUI/Avalonia
namespace Cerberus.Android.Hardware
{
public class CerberusRs485Manager
{
private UsbSerialPort _serialPort;
private UsbDeviceConnection _connection;
private const int BAUD_RATE = 9600; // Стандартная скорость для большинства приборов Цербер (измените, если в софте другая)
public bool ConnectToCerberus(Context context)
{
var usbManager = (UsbManager)context.GetSystemService(Context.UsbService);
var availableDrivers = UsbSerialProber.DefaultProber.FindAllDrivers(usbManager);
if (availableDrivers.Count == 0) return false;
// Берем первый найденный конвертер USB-RS485 (CH340/CP2102)
IUsbSerialDriver driver = availableDrivers[0];
UsbDevice device = driver.Device;
// Запрашиваем у пользователя Samsung разрешение на работу с USB-устройством
if (!usbManager.HasPermission(device)) return false;
_connection = usbManager.OpenDevice(device);
if (_connection == null) return false;
// Открываем порт (обычно у конвертеров RS-485 один порт с индексом 0)
_serialPort = driver.Ports[0];
_serialPort.Open(_connection);
_serialPort.SetParameters(BAUD_RATE, UsbSerialPort.DataBits8, UsbSerialPort.StopBits1, UsbSerialPort.ParityNone);
return true;
}
public void SendCommand(byte[] buffer)
{
if (_serialPort != null)
{
_serialPort.Write(buffer, 2000); // Тайм-аут 2 секунды
}
public byte[] ReadResponse()
{
if (_serialPort == null) return Array.Empty<byte>();
byte[] buffer = new byte[1024];
int numBytesRead = _serialPort.Read(buffer, 2000);
byte[] result = new byte[numBytesRead];
Array.Copy(buffer, result, numBytesRead);
return result;
}
public void Disconnect()
{
_serialPort?.Close();
_connection?.Close();
}

else
{

if (separatorIndex > 0)
{
string key = line.Substring(0, separatorIndex).Trim();
string value = line.Substring(separatorIndex + 1).Trim();
if (!Sections.ContainsKey(currentSection))
{
Sections[currentSection] = new Dictionary<string, string>(StringComparer.OrdinalIgnoreCase);
}
Sections[currentSection][key] = value;
}

Ниже представлена полная конфигурация сетевого сокета для Android-версии. Она позволит вашему телефону Samsung M12 опрашивать прибор GP-4 по сети через Wi-Fi, дублируя поведение ПК-версии.
csharp
using System;
using System.Net;
using System.Net.Sockets;
using System.Threading.Tasks;
namespace Cerberus.Android.Network
{
public class CerberusNetworkManager
{
private TcpClient _tcpClient;
private NetworkStream _networkStream;
private const string DEFAULT_IP = "192.168.0.200"; // IP-адрес прибора с вашего первого скриншота
private const int DEFAULT_PORT = 5000; // Типичный порт для преобразователей RS485-Ethernet
public async Task<bool> ConnectToDeviceAsync(string ipAddress = DEFAULT_IP, int port = DEFAULT_PORT)
{
try
{
_tcpClient = new TcpClient();
// Установка таймаута подключения 3 секунды
var connectTask = _tcpClient.ConnectAsync(ipAddress, port);
var delayTask = Task.Delay(3000);
var completedTask = await Task.WhenAny(connectTask, delayTask);
if (completedTask == delayTask || !_tcpClient.Connected)
{
Disconnect();
return false;
}
_networkStream = _tcpClient.GetStream();
return true;
}
catch (Exception)
{
Disconnect();
return false;
}
public async Task SendCommandAsync(byte[] command)
{
if (_networkStream != null && _tcpClient.Connected)
{
await _networkStream.WriteAsync(command, 0, command.Length);
await _networkStream.FlushAsync();
}
public async Task<byte[]> ReadResponseAsync()
{
if (_networkStream == null || !_tcpClient.Connected) return Array.Empty<byte>();
byte[] buffer = new byte[1024];
if (_networkStream.DataAvailable)
{
int bytesRead = await _networkStream.ReadAsync(buffer, 0, buffer.Length);
byte[] result = new byte[bytesRead];
Array.Copy(buffer, result, bytesRead);
return result;
}
return Array.Empty<byte>();
}
public void Disconnect()
{
_networkStream?.Close();
_tcpClient?.Close();
}

using System;
using System.Collections.Generic;
namespace Cerberus.Android.Model
{
// Режимы работы физического порта М1
public enum PortM1Mode : byte
{
Disabled = 0,     // Отключен
RadioBobs = 1,    // Радио Брелоки
LedIndicator = 2  // Светодиодный индикатор
}
// Финальный агрегатор всей конфигурации прибора Цербер04 (Карта файла .c4)
public class CerberusFullConfiguration
{
// 1. Системные данные из шапки софта
public string DeviceModel { get; set; } = "ЦЕРБЕР RP4 rev3.0-3.4";
public string SerialNumber { get; set; } = "N/A";
public string FirmwareVersion { get; set; } = "N/A";
public byte BusAddress { get; set; } = 0;
// 2. Все вкладки конфигуратора, собранные вместе
public List<CerberusPartition> Partitions { get; set; } = new List<CerberusPartition>();
public List<CerberusZone> Zones { get; set; } = new List<CerberusZone>();
public List<CerberusOutput> Outputs { get; set; } = new List<CerberusOutput>();
public List<CerberusUser> Users { get; set; } = new List<CerberusUser>();
public CerberusWirelessRepository WirelessRepository { get; set; } = new CerberusWirelessRepository();
public CerberusRadioNetworkSettings RadioNetwork { get; set; } = new CerberusRadioNetworkSettings();
public CerberusRs485Settings Rs485 { get; set; } = new CerberusRs485Settings();
public CerberusAdditionalSettings Additional { get; set; } = new CerberusAdditionalSettings();
// Финальный параметр: Режим порта М1
public PortM1Mode M1Mode { get; set; } = PortM1Mode.Disabled;
public CerberusFullConfiguration()
{
InitializeDefaultStructure();
}
private void InitializeDefaultStructure()
{
// Создаем 8 базовых разделов (как в оригинальном софте)
for (int i = 1; i <= 8; i++)
Partitions.Add(new CerberusPartition { Id = i });
// Создаем 12 шлейфов (8 на плате + 4 клавиатурных)
for (int i = 1; i <= 12; i++)
{
Zones.Add(new CerberusZone
{
Id = i,
ResistanceInfo = i <= 4 ? "10кОм" : (i <= 8 ? "20кОм" : "Клавиатура")
});
}
// Создаем 4 независимых выхода 12В
for (int i = 1; i <= 4; i++)
Outputs.Add(new CerberusOutput { Id = i });
// Создаем 250 ячеек пользователей под ТМ-ключи и PIN-коды
for (int i = 1; i <= 250; i++)
Users.Add(new CerberusUser { SlotNumber = i });
}
// Экспорт всей конфигурации в единый бинарный массив для записи в прибор или файл .c4
public byte[] ExportToBinaryBlob()
{
List<byte> blob = new List<byte>();
// Пример упаковки: пишем байты заголовка, затем последовательно все структуры
blob.Add(BusAddress);
blob.Add((byte)M1Mode);
blob.AddRange(Rs485.GetBytes());
blob.AddRange(Additional.GetBytes());
blob.AddRange(RadioNetwork.GetBytes());
foreach (var zone in Zones) blob.Add(zone.GetFlagsByte());
foreach (var outItem in Outputs) blob.AddRange(outItem.GetBytes());
return blob.ToArray();
}

Общий принцип Пядевой системы мер базируется на масштабируемой гармонии человеческого тела и природных циклов, где основной единицей является Пядь (17,78 см). Все остальные меры делятся или умножаются строго в пропорциях, кратных древней двоичной и шестнадцатеричной системам счисления:
Малые меры получаются делением Пяди на части (доли), кратные 2, 4, 16, 256 и 4096 (Вершок, Ноготь, Линия, Волос) для микровзаимодействий.
Средние меры расширяют Пядь для бытового строительства через кратные целые коэффициенты (Стопа — 2ћ, Локоть — 3ћ, Аршин — 4ћ, Сажень — 12ћ).
Большие и Космические меры уходят в макрокосмос для расчета расстояний между планетами и землями (Даль, Светлая даль, Дальняя даль), связывая человека со Вселенной единой масштабной линейкой.
