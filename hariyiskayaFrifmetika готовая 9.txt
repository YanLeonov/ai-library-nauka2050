    <script>
        let activeLanguage = 'modern';
        let selectedElement = 'res'; 

        const translationMatrix = {
            modern: {
                mainTitle: 'Х`Арийская арифметика',
                subTitle: 'Триадное умножение "ЖДЫ"',
                labelModern: 'Современный стандарт',
                labelSlavic: 'Древне Славянский стиль',
                titleFlatLeft: 'Плоскостное умножение "ЖДЫ"', 
                titleFlatRight: 'одна мера',                  
                descSingle: 'Единичная мера (структурная плотность)',
                descMatrix: 'Раскрытие одной меры в плоскость 3х3 (9 структур)',
                val1: '1', val2: '1', valRes: '9',
                lbl1Bot: '', lbl2Bot: '', lblResBot: ''
            },
            slavic: {
                mainTitle: 'Х`Ариіская ариѳметика', 
                subTitle: 'Тріадное умноженіе "ЖДЫ"', 
                labelModern: 'Соврѣменный стандартъ',
                labelSlavic: 'Дрѣвлѣ-славянскiй стиль',
                titleFlatLeft: 'Плоскостное умноженіе "ЖДЫ"', 
                titleFlatRight: 'едина мера',                  
                descSingle: 'Единичная мѣра (структурная плотность)',
                descMatrix: 'Раскрытіе одной мѣры въ плоскость 3х3 (9 структуръ)',
                val1: 'a', val2: 'a', valRes: 'ѳ',
                lbl1Bot: 'Азъ', lbl2Bot: 'Азъ', lblResBot: 'Фита'
            }
        };

        function handleStyleToggle(style) {
            activeLanguage = style;
            syncInterfaceText();
            renderEngine();
        }

        function handleElementClick(elementId) {
            selectedElement = elementId;
            document.getElementById('cellN1').classList.toggle('active-target', elementId === 'n1');
            document.getElementById('cellN2').classList.toggle('active-target', elementId === 'n2');
            document.getElementById('cellRes').classList.toggle('active-target', elementId === 'res');
            syncInterfaceText();
            renderEngine();
        }

        function syncInterfaceText() {
            const dictionary = translationMatrix[activeLanguage];
            document.getElementById('mainAppTitle').innerText = dictionary.mainTitle;
            document.getElementById('subSystemTitle').innerText = dictionary.subTitle;
            document.getElementById('labelModern').innerText = dictionary.labelModern;
            document.getElementById('labelSlavic').innerText = dictionary.labelSlavic;
            
            document.getElementById('cellN1').innerText = dictionary.val1;
            document.getElementById('cellN2').innerText = dictionary.val2;
            document.getElementById('cellRes').innerText = dictionary.valRes;
            
            document.getElementById('labelN1Bottom').innerText = dictionary.lbl1Bot;
            document.getElementById('labelN2Bottom').innerText = dictionary.lbl2Bot;
            document.getElementById('labelResBottom').innerText = dictionary.lblResBot;
            
            document.getElementById('panelTitleLeft').innerText = dictionary.titleFlatLeft;
            document.getElementById('panelTitleRight').innerText = dictionary.titleFlatRight;
            
            document.getElementById('panelDescriptionText').innerText = (selectedElement === 'n1' || selectedElement === 'n2') ? dictionary.descSingle : dictionary.descMatrix;
        }

        function renderEngine() {
            const container = document.getElementById('canvasDisplayNode');
            container.innerHTML = '';
            
            const canvas = document.createElement('canvas');
            // Оптимальный размер холста под равномерный шаг gap
            canvas.width = 500;
            canvas.height = 340;
            container.appendChild(canvas);
            const ctx = canvas.getContext('2d');

            let mouseX = canvas.width / 2;
            let mouseY = canvas.height / 2;

            let spheres = [];
            const isSingle = (selectedElement === 'n1' || selectedElement === 'n2');

            if (isSingle) {
                spheres.push({ x: canvas.width / 2, y: canvas.height / 2, r: 110 });
            } else {
                const cx = canvas.width / 2;
                const cy = canvas.height / 2;
                
                const startX = cx - 100;
                const startY = cy - 100;
                for(let r=0; r<3; r++) {
                    for(let c=0; c<3; c++) {
                        spheres.push({ x: startX + c*100, y: startY + r*100, r: 50 });
                    }
                }
            }

            function draw() {
                ctx.clearRect(0, 0, canvas.width, canvas.height);
                
                spheres.forEach(s => {
                    let dx = mouseX - s.x;
                    let dy = mouseY - s.y;
                    let dist = Math.sqrt(dx*dx + dy*dy) || 1;
                    
                    let fx = s.x + (dx / dist) * (s.r * 0.4);
                    let fy = s.y + (dy / dist) * (s.r * 0.4);

                    let grad = ctx.createRadialGradient(fx, fy, s.r * 0.05, s.x, s.y, s.r);
                    grad.addColorStop(0, '#ffffff');
                    grad.addColorStop(0.2, '#fca5a5');
                    grad.addColorStop(1, '#ef4444');

                    ctx.beginPath();
                    ctx.arc(s.x, s.y, s.r, 0, Math.PI * 2);
                    ctx.fillStyle = grad;
                    ctx.fill();
                    ctx.closePath();
                });
            }

            canvas.addEventListener('mousemove', (e) => {
                const rect = canvas.getBoundingClientRect();
                mouseX = e.clientX - rect.left;
                mouseY = e.clientY - rect.top;
                draw();
            });

            canvas.addEventListener('mouseleave', () => {
                mouseX = canvas.width / 2;
                mouseY = canvas.height / 2;
                draw();
            });

            draw();
        }

        syncInterfaceText();
        handleElementClick('res');
    </script>
